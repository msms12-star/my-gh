clothing_advisor.py 
def give_clothing_advice(temperature):
    if temperature > 25:
        print("It's hot outside! Wear light clothing.")
    elif 15 <= temperature <= 25:
        print("It's mild outside. A t-shirt and jeans should be fine.")
    elif 5 <= temperature < 15:
        print("It's a bit chilly. Consider a jacket.")
    else:
        print("It's cold! Dress warmly and wear a coat.")

def get_temperature():
    try:
        return float(input("Enter the current temperature in Celsius: "))
    except ValueError:
        print("Please enter a valid number.")
        return get_temperature()

temperature = get_temperature()
give_clothing_advice(temperature)

student_count = int(input("Enter the number of students in the class: "))
student_names = []

for _ in range(student_count):
    name = input("Enter the student's name: ")
    student_names.append(name)

sorted_student_names = sorted(student_names)

print("\nClass Roster in Alphabetical Order:")
for student in sorted_student_names:
    print(student)

import numpy as np

# Create a Nutritional Data Array
nutritional_data = np.array([
    [200, 30, 10],  # Food Item 1 : calories, carbs, proteins
    [300, 50, 20],  # Food Item 2
    [250, 45, 15]   # Food Item 3
])

# Calculate Basic Statistics
average_values = np.mean(nutritional_data, axis=0)
min_values = np.min(nutritional_data, axis=0)
max_values = np.max(nutritional_data, axis=0)

# Display the Results
print("Average Nutritional Values (Calories, Carbs, Proteins): ", average_values)
print("Minimum Nutritional Values (Calories, Carbs, Proteins): ", min_values)
print("Maximum Nutritional Values (Calories, Carbs, Proteins): ", max_values)
